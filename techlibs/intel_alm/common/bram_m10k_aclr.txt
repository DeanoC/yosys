# Standard Cyclone V M10K simple-dual-port inference.  The read output
# register has an active-high asynchronous clear which maps to the physical
# bottom output clear (ACLR1).  Keep this separate from the opt-in TDP styles:
# ramstyle=M10K is the vendor's ordinary SDP spelling.
ram block $__MISTRAL_M10K_ACLR_BYTE_ {
 abits 9;
 width 20;
 byte 10;
 cost 100;
 init any;
 style "m10k";
 port sr "R" {
  clock posedge;
  clken;
  rden;
  rdarst zero;
 }
 port sw "W" {
  clock posedge;
  clken;
  wrbe_separate;
 }
}

ram block $__MISTRAL_M10K_ACLR_ {
 abits 10;
 widths 10 20 40 per_port;
 cost 100;
 init any;
 style "m10k";
 port sr "R" {
  clock posedge;
  clken;
  rden;
  rdarst zero;
 }
 port sw "W" {
  clock posedge;
  clken;
 }
}
